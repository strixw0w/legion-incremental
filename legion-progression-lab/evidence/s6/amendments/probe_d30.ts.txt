// Diagnosis probe (not evidence): reruns S6-E02 task 0's run day by day and prints the first error with its stack.
import { pathToFileURL } from 'node:url';
import { resolve } from 'node:path';
const lab = '<HOME>/Documents/GitHub/legion-incremental/legion-progression-lab';
const imp = async (p: string) => import(pathToFileURL(resolve(lab, p)).href);
const { s6Content } = await imp('src/s6/experiments.ts');
const { LegionSimulation } = await imp('src/engine/legion.ts');
const { pol6 } = await imp('src/policy/w1Policies.ts');
const { s6ProfileSessions } = await imp('src/s6/profiles.ts');
const H = 3_600_000;
const D = 24 * H;
const seed = process.argv[2] ?? 's6-e02#PR-BAL#0';
const profile = process.argv[3] ?? 'PR-BAL';
const policyId = process.argv[4] ?? 'POL6-BAL';
const days = Number(process.argv[5] ?? 30);
const sessions = s6ProfileSessions(profile, days);
const sim = new LegionSimulation({ content: s6Content(), seedRoot: seed, policy: pol6(policyId), sessions: [...sessions], legion: true, industry: true, permanent: true, keepLog: 0, hashLog: false, ledger: 'bounded', identity: 'l2', offline: {}, maxEvents: 1_000_000_000 });
let last = '';
sim.addObserver((e: any) => { last = `${e.kind} t=${e.t}`; });
const t0 = Date.now();
for (let w = D; w <= days * D; w += D) {
  try {
    sim.runToWall(w);
  } catch (err) {
    const e = err as Error;
    console.log(`ERROR at wall day ${w / D} (game ${(sim.state.tMs / H).toFixed(2)} h, last event ${last}): ${e.message}`);
    console.log(e.stack?.split('\n').slice(0, 14).join('\n'));
    console.log(`hero level ${sim.state.hero.level}, max level ${Math.max(...sim.state.legion.chars.map((c: any) => c.level))}, table top ${sim.content.table.length + 1}`);
    process.exit(0);
  }
  console.log(`day ${w / D} ok (${((Date.now() - t0) / 1000).toFixed(0)} s) hero level ${sim.state.hero.level} zones ${sim.state.zones.filter((z: any) => z.accessUnlocked).length} world ${sim.state.boss.completed} events ${sim.counters.eventsProcessed}`);
}
console.log('no error');
